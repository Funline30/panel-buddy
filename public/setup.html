Firestore security rules

Firebase console → Firestore Database → Rules tab → select everything in the box and delete it → paste this → Publish.

rules_version = '2';

// AmpMap - Firestore security rules.
// Each signed-in user can read and write ONLY their own data under users/{their uid}/.
// Nobody (signed in or not) can read anyone else's data.
// Exception: shares/{token} - a read-only panel directory snapshot the owner chose to
// share. Anyone holding the (random, 24+ character) link can open that one document;
// nobody can list or search shares, and only the owner can change or delete it.
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }

    function knownCollection(col) {
      return col in ['homes', 'panels', 'breakers', 'devices', 'photos', 'customers', 'settings'];
    }

    match /users/{uid}/{col}/{docId} {
      allow read, delete: if isOwner(uid) && knownCollection(col);
      allow create, update: if isOwner(uid)
        && knownCollection(col)
        && (col != 'photos'
            || (request.resource.data.data is string
                && request.resource.data.data.size() < 900000));
    }

    match /shares/{token} {
      allow get: if true;
      allow list: if false;
      allow create: if signedIn()
        && token.size() >= 24
        && request.resource.data.owner == request.auth.uid;
      allow update: if signedIn()
        && resource.data.owner == request.auth.uid
        && request.resource.data.owner == request.auth.uid;
      allow delete: if signedIn() && resource.data.owner == request.auth.uid;
    }

    // Lets someone who scans a panel's QR sticker find that panel's share link
    // (only exists while the owner has sharing turned on).
    match /sharedPanels/{panelId} {
      allow get: if true;
      allow list: if false;
      allow create: if signedIn() && request.resource.data.owner == request.auth.uid;
      allow update: if signedIn()
        && resource.data.owner == request.auth.uid
        && request.resource.data.owner == request.auth.uid;
      allow delete: if signedIn() && resource.data.owner == request.auth.uid;
    }

    // Everything else is closed.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

What it does: each signed-in person can read and write only their own data (under users/their-id/). The only exception is a read-only panel directory someone chooses to share: anyone with that exact link can view it, nobody can list or search them, and only the owner can change or delete it. Everything else is locked.
